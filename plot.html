treeio.plot

Phylogenetic tree plotting (matplotlib backend).

Trees are drawn onto a real matplotlib.axes.Axes using ax.plot / ax.scatter / ax.text / LineCollection, so every matplotlib feature (subplots, legends, colormaps, limits, savefig, animations) works. The renderer is object-oriented and matplotlib-compatible.

Layouts

  • rectangular / roundrect -- L-shaped / rounded phylogram
  • circular / fan -- radial / partial-sector tree
  • radial -- equal-angle ring dendrogram
  • slanted -- diagonal phylogram
  • unrooted -- equal-angle circular layout
  • time -- time-scaled tree (tips left, root right)

Examples

>>> import matplotlib.pyplot as plt
>>> from treeio import render, draw_tree, treeplot
>>> tree = read("data/animals.nwk")
>>> ax = render(tree, backend="mpl", layout="circular", tip_labels=True)
>>> plt.savefig("tree.png")

118def tree_coords(tree: Tree, layout: str = "rectangular", reverse_x: bool = False) -> Dict[Tree, Tuple[float, float]]:
119    """Return ``{node: (x, y)}`` in data coordinates for ``layout``."""
120    if layout in _LAYOUT_REGISTRY:
121        return dict(_LAYOUT_REGISTRY[layout](tree))
122    use_depth = layout == "radial"
123    xpos = _x_positions(tree, reverse_x=reverse_x, use_depth=use_depth)
124    tips = tree.get_tips()
125    tipy = {tip: float(i) for i, tip in enumerate(tips)}
126    coords: Dict[Tree, Tuple[float, float]] = {}
127    for node in tree.traverse("postorder"):
128        if not node.children:
129            y = tipy[node]
130        else:
131            y = sum(coords[c][1] for c in node.children) / len(node.children)
132        coords[node] = (xpos[node], y)
133    if layout in ("circular", "fan", "radial", "unrooted"):
134        return _polarize(coords, tree, layout)
135    return coords

138def _x_positions(tree: Tree, reverse_x: bool, use_depth: bool) -> Dict[Tree, float]:
139    if use_depth:
140        def depth(node):
141            d = 0
142            cur = node
143            while cur is not None and cur.parent is not None:
144                d += 1
145                cur = cur.parent
146            return float(d)
147        out = {n: depth(n) for n in tree.traverse("preorder")}
148    else:
149        has_len = any(t.branch_length for t in tree.traverse("preorder") if t.branch_length)
150        if has_len:
151            out = {}
152            stack = [(tree, 0.0)]
153            while stack:
154                node, x = stack.pop()
155                out[node] = x
156                for c in node._children:
157                    d = c.branch_length if c.branch_length is not None else 0.0
158                    stack.append((c, x + d))
159        else:
160            def depth(node):
161                d = 0
162                cur = node
163                while cur is not None and cur._parent is not None:
164                    d += 1
165                    cur = cur._parent
166                return float(d)
167            out = {n: depth(n) for n in tree.traverse("preorder")}
168    if reverse_x:
169        mx = max(out.values(), default=0.0) or 1.0
170        out = {n: mx - x for n, x in out.items()}
171    return out

174def _polarize(coords: Dict[Tree, Tuple[float, float]], tree: Tree, layout: str):
175    n = tree.nleaves or 1
176    xmax = max((c[0] for c in coords.values()), default=0.0) or 1.0
177    out = {}
178    if layout == "fan":
179        sweep = math.pi
180        start = math.pi / 2 - sweep / 2
181        for node, (x, y) in coords.items():
182            r = x / xmax
183            theta = start + (y / n) * sweep
184            out[node] = (r * math.cos(theta), r * math.sin(theta))
185        return out
186    for node, (x, y) in coords.items():
187        r = x / xmax
188        theta = ((y / n) * 2 * math.pi - math.pi / 2) if layout != "unrooted" else ((y / n) * 2 * math.pi)
189        out[node] = (r * math.cos(theta), r * math.sin(theta))
190    return out

193def edge_segments(tree: Tree, coords=None, layout: str = "rectangular") -> List[List[Tuple[float, float]]]:
194    """Return a list of polylines (each a list of points) for every edge."""
195    if coords is None:
196        coords = tree_coords(tree, layout)
197def edge_segments(tree: Tree, coords=None, layout: str = "rectangular") -> List[List[Tuple[float, float]]]:
198    """Return a list of polylines (each a list of points) for every edge."""
199    if coords is None:
200        coords = tree_coords(tree, layout)
201    straight = layout in ("circular", "fan", "radial", "unrooted", "slanted") or layout in _LAYOUT_REGISTRY
202    segs = []
203    for node in tree.traverse("preorder"):
204        if node.parent is None:
205            continue
206        xp, yp = coords[node.parent]
207        xc, yc = coords[node]
208        if straight:
209            segs.append([(xp, yp), (xc, yc)])
210        else:
211            segs.append([(xp, yp), (xp, yc), (xc, yc)])
212    return segs

242def draw_tree(
243    tree: Tree,
244    ax=None,
245    layout: str = "rectangular",
246    tip_labels: bool = True,
247    tip_points: bool = True,
248    node_points: bool = False,
249    node_support: bool = False,
250    tip_colors: Optional[Dict[str, str]] = None,
251    tip_color_field: str = None,
252    show_tip_legend: bool = True,
253    edge_color: str = "#333333",
254    edge_width: float = 1.2,
255    label_size: int = 10,
256    label_radial_offset: float = 0.04,
257    axis_label_size: int = 9,
258    tick_size: int = 8,
259    reverse_x: bool = False,
260    branch_color_field: str = None,
261    cmap: str = "viridis",
262    theme: str = "clean",
263    show_colorbar: bool = False,
264    colorbar_label: str = "value",
265    **kwargs,
266):
267    """Draw ``tree`` onto ``ax`` (creating one if ``ax`` is None) and return it.
268
269    Uses real ``ax`` primitives (``LineCollection`` + ``scatter`` + ``text``)
270    so the returned axes is fully matplotlib-compatible.  Edges are batched
271    into a single collection and all tips into one ``scatter``, so very large
272    trees render quickly.
273
274    ``branch_color_field`` colours each branch by a per-node value and adds a
275    continuous colourbar if ``show_colorbar`` is true; ``theme`` applies a
276    :func:`tree_theme` style.
277    """
278    plt = _mpl()
279    from matplotlib.collections import LineCollection
280    polar = layout in ("circular", "fan", "radial", "unrooted")
281    if ax is None:
282        figsize = kwargs.pop("figsize", None)
283        w, h = _figsize(tree, layout)
284        ax = plt.subplots(figsize=figsize or (w / 80.0, h / 80.0),
285                          subplot_kw={"aspect": "equal"} if polar else None)[1]
286    coords = tree_coords(tree, layout=layout, reverse_x=reverse_x)
287    segments = edge_segments(tree, coords, layout=layout)
288
289    mappable = None
290    if branch_color_field is not None:
291        import numpy as np
292        import matplotlib
293        vals = [c.get_data(branch_color_field) for c in tree.traverse("preorder") if c.parent is not None]
294        arr = np.asarray([np.nan if v is None else float(v) for v in vals], dtype=float)
295        if np.all(np.isnan(arr)):
296            lc = LineCollection(segments, colors=edge_color, linewidths=edge_width,
297                                capstyle="round", antialiaseds=True, zorder=1)
298        else:
299            lo = float(np.nanmin(arr)); hi = float(np.nanmax(arr))
300            if not np.isfinite(lo) or lo == hi:
301                lo, hi = 0.0, 1.0
302            norm = matplotlib.colors.Normalize(vmin=lo, vmax=hi)
303            lc = LineCollection(segments, array=arr, cmap=matplotlib.colormaps.get_cmap(cmap),
304                                norm=norm, linewidths=edge_width, capstyle="round", zorder=1)
305            mappable = lc
306    else:
307        lc = LineCollection(segments, colors=edge_color, linewidths=edge_width,
308                            capstyle="round", antialiaseds=True, zorder=1)
309    ax.add_collection(lc)
310
311    tips = tree.get_tips()
312    tip_x = [coords[t][0] for t in tips]
313    tip_y = [coords[t][1] for t in tips]
314    tip_c, tip_mappable, tip_legend = _tip_colors_for(tree, tips, tip_colors, tip_color_field, cmap, edge_color)
315    if tip_points:
316        ax.scatter(tip_x, tip_y, s=18, c=tip_c, zorder=3, edgecolors="white", linewidths=0.4)
317    if tip_labels:
318        if polar:
319            import math
320            n = len(tips) or 1
321            # wrap-around-safe angular crowding drives the radial offset so
322            # dense clusters push their labels outward (avoiding overlap).
323            threshold = (2 * math.pi / n) * 0.6
324            angles = [math.atan2(tip_y[i], tip_x[i]) for i in range(n)]
325            offset_map = {}
326            for i in range(n):
327                a = angles[i]
328                crowd = sum(1 for j in range(n)
329                            if abs((angles[j] - a + math.pi) % (2 * math.pi) - math.pi) < threshold)
330                offset_map[i] = label_radial_offset * (1 + 0.3 * max(0, crowd - 1))
331        else:
332            offset_map = {}
333        for i, (t, x, y, c) in enumerate(zip(tips, tip_x, tip_y, tip_c)):
334            off = offset_map.get(i, label_radial_offset)
335            _write_tip_label(ax, x, y, t.name, polar, c, label_size, offset=off)
336
337    nodes = tree.get_internal_nodes()
338    if node_points:
339        ax.scatter([coords[n][0] for n in nodes], [coords[n][1] for n in nodes], s=12, c=["#999999"], zorder=3)
340    if node_support:
341        for n in nodes:
342            if n.support is not None:
343                x, y = coords[n]
344                ax.text(x, y + 0.05, str(n.support), fontsize=label_size - 2, color="#666666",
345                        ha="center", va="bottom")
346
347    if polar and theme == "void":
348        ax.set_axis_off()
349    elif not polar:
350        tree_theme(ax, style=theme)
351        ax.set_xlabel("branch length" if any(t.branch_length for t in tree.traverse("preorder") if t.branch_length) else "edges",
352                      fontsize=axis_label_size)
353        ax.tick_params(labelsize=tick_size, direction="out")
354        if len(tips) <= 200:
355            ax.set_yticks(list(range(len(tips))))
356            ax.set_yticklabels([])
357    ax.autoscale()
358    # colourbar / legend for the tip field
359    if tip_mappable is not None and show_tip_legend:
360        cb = ax.figure.colorbar(tip_mappable, ax=ax)
361        cb.set_label(tip_color_field, fontsize=9)
362    elif tip_legend and show_tip_legend:
363        import matplotlib.patches as mpatches
364        handles = [mpatches.Patch(color=c, label=str(k)) for k, c in tip_legend]
365        ax.legend(handles=handles, title=tip_color_field, fontsize=8, title_fontsize=9,
366                  loc="upper left", bbox_to_anchor=(1.02, 1.0))
367    if show_colorbar and mappable is not None:
368        cb = ax.figure.colorbar(mappable, ax=ax)
369        if colorbar_label:
370            cb.set_label(colorbar_label)
371        ax._treeio_mappable = mappable
372    return ax

375def _tip_colors_for(tree, tips, tip_colors, tip_color_field, cmap, default):
376    """Return ``(colors, mappable, legend)`` for tip markers.
377
378    If ``tip_color_field`` is given the colour is derived from each tip's value
379    (continuous -> colourbar mappable; categorical -> discrete legend).
380    """
381    import numpy as np
382    import matplotlib
383    if tip_color_field is not None:
384        values = {t.name: t.get_data(tip_color_field) for t in tips}
385        known = {k: v for k, v in values.items() if v is not None}
386        if known and all(isinstance(v, (int, float)) for v in known.values()):
387            arr = np.array([float(values[t.name]) if values[t.name] is not None else np.nan for t in tips])
388            lo = float(np.nanmin(arr)); hi = float(np.nanmax(arr))
389            if not np.isfinite(lo) or lo == hi:
390                lo, hi = 0.0, 1.0
391            norm = matplotlib.colors.Normalize(vmin=lo, vmax=hi)
392            cobj = matplotlib.colormaps.get_cmap(cmap)
393            colors = cobj(norm(arr))
394            mappable = matplotlib.cm.ScalarMappable(norm=norm, cmap=cobj)
395            return colors, mappable, None
396        cats = sorted(set(known.values()), key=str)
397        pal = named_palette([str(c) for c in cats])
398        colors = [pal.get(str(values[t.name]), default) for t in tips]
399        return colors, None, [(c, pal[str(c)]) for c in cats]
400    return [(tip_colors or {}).get(t.name, default) for t in tips], None, None

403def _write_tip_label(ax, x, y, text, polar, color, size, offset=0.04):
404    """Place a tip label; radial layouts rotate it so labels do not overlap."""
405    if polar:
406        import math
407        theta = math.degrees(math.atan2(y, x))
408        lx = x + math.cos(math.radians(theta)) * offset
409        ly = y + math.sin(math.radians(theta)) * offset
410        rot = theta
411        if 90 < theta < 270:
412            rot = theta - 180
413            ha = "right"
414        else:
415            ha = "left"
416        return ax.text(lx, ly, text, rotation=rot, rotation_mode="anchor", ha=ha, va="center",
417                       fontsize=size, color=color, zorder=4)
418    return ax.text(x, y, text, fontsize=size, color=color, va="center", ha="left", zorder=4)

424class TreePlotter:
425    """An OO, matplotlib-compatible tree plotter.
426
427    Holds a real :class:`matplotlib.axes.Axes` and exposes marker methods that
428    delegate to the matching ``ax`` method (``.plot_edges()``,
429    ``.scatter_tips()``, ``.scatter_nodes()``, ``.plot_tip_labels()``,
430    ``.plot_node_labels()``, ``.legend()``, ``.savefig()``)."""
431
432    def __init__(self, tree: Tree, ax=None, layout: str = "rectangular",
433                 edge_color: str = "#333333", edge_width: float = 1.2, **kw):
434        self.tree = tree
435        self.layout = layout
436        self.edge_color = edge_color
437        self.edge_width = edge_width
438        self.reverse_x = kw.pop("reverse_x", False)
439        self.coords = tree_coords(tree, layout=layout, reverse_x=self.reverse_x)
440        self._ax = _new_axes(tree, layout, self.reverse_x, ax=ax, **kw)
441
442    @property
443    def ax(self):
444        """The underlying matplotlib axes (use any ``ax`` method freely)."""
445        return self._ax
446
447    def scatter_tips(self, size: float = 18, color=None, aes: str = None, cmap: str = "viridis", **kw):
448        """Mark tips (``ax.scatter``).  ``aes`` colours by a per-tip field."""
449        tips = self.tree.get_tips()
450        c, _, _ = _tip_colors_for(self.tree, tips, None, aes, cmap, self.edge_color)
451        return self._ax.scatter([self.coords[t][0] for t in tips],
452                                [self.coords[t][1] for t in tips],
453                                s=size, c=color or c, **kw)
454
455    def scatter_nodes(self, size: float = 12, color="#999999", **kw):
456        """Mark internal nodes (``ax.scatter``)."""
457        nodes = self.tree.get_internal_nodes()
458        return self._ax.scatter([self.coords[n][0] for n in nodes],
459                                [self.coords[n][1] for n in nodes],
460                                s=size, c=color, **kw)
461
462    def plot_tip_labels(self, size: int = 10, aes: str = None, color=None,
463                        radial_offset: float = 0.04, **kw):
464        """Write tip labels; radial layouts rotate them (collision avoidance).
465
466        ``aes`` colours the labels by a per-tip field."""
467        polar = self.layout in ("circular", "fan", "radial", "unrooted")
468        c, _, _ = _tip_colors_for(self.tree, self.tree.get_tips(), None, aes, "viridis", self.edge_color)
469        out = []
470        for i, tip in enumerate(self.tree.get_tips()):
471            x, y = self.coords[tip]
472            col = color or (c[i] if isinstance(c[i], str) else self.edge_color)
473            out.append(_write_tip_label(self._ax, x, y, tip.name, polar, col, size, offset=radial_offset))
474        return out
475
476    def plot_node_labels(self, size: int = 8, color="#666666", show_support=True, **kw):
477        """Write internal-node labels / support values (``ax.text``)."""
478        out = []
479        for node in self.tree.get_internal_nodes():
480            label = node.name
481            if (label in (None, "", "unknown")) and show_support and node.support is not None:
482                label = str(node.support)
483            if label in (None, "", "unknown"):
484                continue
485            x, y = self.coords[node]
486            out.append(self._ax.text(x, y, label, fontsize=size, color=color, ha="center", va="center", **kw))
487        return out

489    def plot_edges(self, aes: str = None, color=None, width=None, cmap: str = "viridis", norm=None):
490        """Draw the branches.
491
492        If ``aes`` names a per-node field, each edge is coloured by the child
493        node's value (a continuous colour scale); call :meth:`add_colorbar` to
494        add a colourbar.  Otherwise a single ``LineCollection`` is used.
495        """
496        from matplotlib.collections import LineCollection
497        segments = edge_segments(self.tree, self.coords, self.layout)
498        if aes is not None:
499            import numpy as np
500            import matplotlib
501            vals = [
502                c.get_data(aes)
503                for c in self.tree.traverse("preorder")
504                if c.parent is not None

671def _nice_unit(raw: float) -> float:
672    """A round (1/2/5 * 10^k) value near ``raw``."""
673    if raw <= 0:
674        return 1.0
675    import math as _m
676    exp = _m.floor(_m.log10(raw))
677    base = 10.0 ** exp
678    for mult in (1, 2, 5, 10):
679        if base * mult >= raw:
680            return base * mult
681    return base * 10

684def add_scalebar(ax, coords, unit: float = None, label: str = "branch length",
685                 loc: Tuple[float, float] = None):
686    """Draw a branch-unit scale bar in data coordinates.
687
688    ``coords`` is the node -> ``(x, y)`` mapping from :func:`tree_coords`.
689    """
690    import numpy as np
691    xs = [c[0] for c in coords.values()]
692    ys = [c[1] for c in coords.values()]
693    lo_x, hi_x = min(xs), max(xs)
694    lo_y, hi_y = min(ys), max(ys)
695    span = (hi_x - lo_x) or 1.0
696    if unit is None:
697        unit = _nice_unit(span / 5.0)
698    if loc is None:
699        x0 = lo_x + 0.02 * span
700        y0 = hi_y + 0.04 * (hi_y - lo_y + 1.0)
701    else:
702        x0, y0 = loc

704    h = 0.02 * (hi_y - lo_y + 1.0)
705    ax.plot([x0, x0], [y0 - h, y0 + h], color="#333333", lw=1.2)
706    ax.plot([x0 + unit, x0 + unit], [y0 - h, y0 + h], color="#333333", lw=1.2)

708            ha="center", va="bottom", fontsize=8, color="#333333")
